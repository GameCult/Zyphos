% ---------- The hand ----------
% Zapf Chancery through NFSS (psnfss `pzc`): LuaTeX's fontspec cannot load the Type 1 file directly.
% It has one weight, so strong and soft are colour, not weight.
\newcommand{\hand}{\fontencoding{T1}\fontfamily{pzc}\fontseries{m}\fontshape{n}\selectfont}
\newcommand{\labelfont}{\hand}
\newcommand{\thinfont}{\hand}
\newcommand{\speakfont}{\hand}
\newcommand{\numfont}{\hand}

% ---------- Ink primitives, called by the generated layer ----------
% A flat ink fill with a hairline edge: a pen stroke, no glow. #2 is the percent of ink kept.
\newcommand{\neonfill}[3]{\path[fill=#1!#2!black!6, draw=#1!85!black, line width=0.12pt, line join=round] #3;}
\newcommand{\neondot}[4]{%
  \fill[#1!70!white, opacity=0.30] (#2,#3) circle ({1.7*#4});
  \fill[#1!85!black] (#2,#3) circle (#4);}
\newcommand{\petalwash}[2]{\fill[#1, opacity=0.13] #2;}
% A ruled frame: parchment panel, a heavy and a hairline rule.
\newcommand{\neonframe}[2]{%
  \fill[panelbg, opacity=0.92] #2;
  \draw[#1!80!black, line width=0.7pt] #2;
  \draw[#1!80!black, line width=0.2pt, opacity=0.7] #2;}

% ---------- Text atoms ----------
% Defined in the preamble (a parameter macro inside a panel breaks tcolorbox fit).
\newcommand{\head}[1]{{\hand\color{accent}#1}\par\vspace{0.25em}}
\newcommand{\sub}[1]{\par\vspace{0.25em}{\hand\color{muted}#1}\par}
\DeclareRobustCommand{\code}[1]{\leavevmode{\color{neonsky}#1}}
\DeclareRobustCommand{\strong}[1]{\leavevmode{\color{heading}#1}}
\DeclareRobustCommand{\soft}[1]{\leavevmode{\color{heading}#1}}
\DeclareRobustCommand{\faint}[1]{\leavevmode{\color{muted}#1}}
\newcommand{\li}[3]{\par{\leftskip=\dimexpr #1em - 1em\relax\hangindent=1em\hangafter=1\noindent\makebox[1em][l]{\color{accent}#2}#3\par}}
\newcommand{\quoted}[1]{\par{\leftskip=0.8em\color{heading}#1\par}}
\newcommand{\codeline}[1]{\par{\color{neonsky}\hangindent=1em\hangafter=1\noindent#1\par}}
\newcommand{\hairline}{\par\vspace{0.2em}{\color{muted}\rule{\linewidth}{0.2pt}}\par\vspace{0.2em}}
\newcommand{\cell}[1]{\raggedright\arraybackslash#1}

% ---------- One panel (as the skill's) ----------
\newcommand{\panel}[6]{%
  \node[anchor=south west, inner sep=0pt] at (#1mm,#2mm) {%
    \begin{tcolorbox}[blank, fit, width=#3mm, height=#4mm,
      left=3.1mm, right=3.1mm, top=3mm, bottom=2.6mm,
      fontupper=\hand\color{body}\linespread{1.04}\selectfont,
      before upper={\colorlet{accent}{#5}\parindent0pt\parskip0.14em\raggedright}]
    #6
    \end{tcolorbox}};}
